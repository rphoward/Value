(def-ref modes
  (linked-from protocol-2-evaluation-references)

  (section writing-modes-archetypes
    (purpose "First task: identify which mode governs the work — mode determines what good looks like")
    (mode argumentation
      (core-question "Is this claim defensible and persuasive?")
      (typical-forms "academic papers, op-eds, legal briefs, speeches"))
    (mode narrative
      (core-question "Does this carry the reader through a transformation?")
      (typical-forms "fiction, memoir, creative nonfiction, poetry"))
    (mode exposition
      (core-question "Does the reader understand this now?")
      (typical-forms "textbooks, tutorials, API docs, explainers"))
    (mode inquiry
      (core-question "Has this uncovered or examined something true?")
      (typical-forms "research, investigative journalism, philosophical essays"))
    (mode documentation
      (core-question "Is this precise, complete, and accountable?")
      (typical-forms "contracts, clinical notes, specs, regulations"))))
